# ============================================
# API Backend
# ============================================
# URL da API Backend
# Exemplo para desenvolvimento local: http://localhost:8000
# Exemplo para produção: https://api.aiexx.com
VITE_API_URL=http://localhost:8000

# ============================================
# Informações da Aplicação
# ============================================
# Nome da aplicação (usado em alguns lugares da UI)
VITE_APP_NAME=AIEXX 3D

# Versão da aplicação
VITE_APP_VERSION=1.0.0

# URL da aplicação (para OAuth callback)
# Exemplo para desenvolvimento local: http://localhost:5173
# Exemplo para produção: https://seu-app.vercel.app
# ⚠️ IMPORTANTE: Esta URL deve ser configurada também no Supabase (Authorized URLs)
VITE_APP_URL=http://localhost:5173

# ============================================
# Supabase Configuration
# ============================================
# 
# COMO OBTER AS CREDENCIAIS DO SUPABASE:
# 1. Acesse: https://app.supabase.com
# 2. Faça login ou crie uma conta
# 3. Crie um novo projeto ou selecione um existente
# 4. Vá em: Settings > API
# 5. Copie a "Project URL" e cole abaixo em VITE_SUPABASE_URL
# 6. Copie a "anon public" key e cole abaixo em VITE_SUPABASE_ANON_KEY
#
# ⚠️ ATENÇÃO: A chave anon será exposta ao cliente (browser)
# Por isso, use apenas a chave "anon public", nunca a "service_role" key
# ============================================

# URL do projeto Supabase
# Formato: https://[seu-projeto-id].supabase.co
# Exemplo: https://grpxuporwqdyckkyhlcx.supabase.co
# PREENCHA COM SUA URL DO SUPABASE:
VITE_SUPABASE_URL=https://grpxuporwqdyckkyhlcx.supabase.co

# Chave pública (anon key) do Supabase
# Esta é a chave "anon public" do seu projeto
# ⚠️ ATENÇÃO: Esta chave será exposta ao cliente (browser)
# Formato: eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...
# PREENCHA COM SUA CHAVE ANON DO SUPABASE:
VITE_SUPABASE_ANON_KEY=your_supabase_anon_key_here

# ============================================
# Supabase - Autenticação Google (Desenvolvimento Local)
# ============================================
# 
# Para usar autenticação Google no desenvolvimento local com Supabase CLI:
# 1. Crie um projeto no Google Cloud Console: https://console.cloud.google.com
# 2. Vá em: APIs & Services > Credentials
# 3. Clique em "Create Credentials" > "OAuth client ID"
# 4. Configure:
#    - Application type: Web application
#    - Authorized redirect URIs: http://localhost:54321/auth/v1/callback
# 5. Copie o "Client ID" e "Client Secret"
# 6. Configure o Client ID no arquivo web/supabase/config.toml
# 7. Adicione o Client Secret abaixo:
#
# ⚠️ ATENÇÃO: Esta variável é usada apenas para desenvolvimento local
# ⚠️ NÃO será exposta ao cliente, mas NUNCA commite este arquivo
# ============================================

# Client Secret do Google OAuth (para desenvolvimento local)
# Obtenha em: https://console.cloud.google.com/apis/credentials
# SUPABASE_AUTH_EXTERNAL_GOOGLE_CLIENT_SECRET=your_google_client_secret_here
